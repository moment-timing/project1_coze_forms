# 项目说明
此智能体用于生成电商大盘数据，在你们数据源获取时对数据做个聚合处理，包括年度、季度、月度和品牌4个方向做聚合，转成一下json格式后请求coze API即可输出报告。
{
  "meta": {  // ===== 元信息：本次统计的全局描述 =====
    "stat_time": "2026-09-10 14:20:43",  // 统计时间，格式：年-月-日 时:分:秒
    "platform_list": ["抖音", "天猫", "京东"],  // 参与统计的平台列表
    "by_brand": false,  // 是否按品牌维度细分：true-按品牌细分，false-不细分
    "by_platform_brand": true,  // 是否按“平台+品牌”聚合：true-是，false-否
    "summary_scope": "all",  // 汇总范围：all-跨全部平台所有类目，platform-每个平台内部跨类目
    "processed_files": 4,  // 成功读取并处理的文件数量
    "skipped_count": 0,  // 跳过（读取失败或空文件）的文件数量
    "shop_name": "浴室角架",  // 用户输入的商品名称
    "data_size": "15.39KB",  // 本次数据体积（可读字符串）
    "data_size_bytes": 15761  // 本次数据体积（字节数）
  },

  "category_list": [  // ===== 类目维度：每个平台下的每个类目独立统计 =====
    {
      "platform": "抖音",  // 数据所属平台，如：抖音/天猫/京东
      "category_name": "家庭/个人清洁工具>卫浴/置物用具>浴室角架",  // 类目完整路径

      "agg_year": [  // ---- 按年聚合统计结果 ----
        {
          "年": "2023",  // 年份
          "销售额(元)": 111051930.0,  // 销售额合计，单位：元

        // ... 每年一条（2023 ~ 2026）
      ],

      "agg_quarter": [  // ---- 按季度聚合统计结果 ----
        {
          "年": "2023",  // 年份
          "季度": "2023Q1",  // 季度标识，格式：YYYYQN
          "销售额(元)": 22048201.0,  // 销售额合计，单位：元

        }
        // ... 每季度一条（2023Q1 ~ 2026Q3）
      ],

      "agg_brand": []  // ---- 类目级品牌聚合，此处为空（品牌数据统一在 platform_brand_summary）----
    }
    // ... 每个平台下每个类目一条（本数据共 4 条）
  ],

  "global_monthly_summary": [  // ===== 全局月度汇总：所有平台所有类目按月合计 =====
    {
      "月": "202301",  // 月份标识，格式：YYYYMM
      "销售额(元)": 85432577.0,  // 该月所有类目销售额合计
      "销量(件)": 2585921,  // 该月所有类目销量合计
      "均价(元)": 33.04  // 该月平均单价 = 总销售额 / 总销量
    }
    // ... 每月一条（2023-01 ~ 2026-07，共 42 条）
  ],

  "platform_brand_summary": [  // ===== 平台×品牌汇总：每个平台下各品牌的销售额/销量/均价 =====
    {
      "platform": "抖音",  // 平台名
      "brand_list": [  // 该平台内的品牌列表（按销售额降序）

          "品牌": "其他 Other",  // 品牌名（含“其他 Other”长尾归类）

    }
    // ... 每个平台一条（抖音 / 天猫 / 京东）

必要的字段有就行，其他字段无关紧要的字段可在数据处理时自己选择添加或删除。
